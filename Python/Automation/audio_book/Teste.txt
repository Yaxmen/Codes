---
- name: START - Identificar plataforma Cisco
  hosts: localhost
  gather_facts: false
  connection: local

  tasks:
    - name: Validar host informado
      ansible.builtin.assert:
        that:
          - target_host is defined
          - target_host | string | trim | length > 0
        fail_msg: "Informe target_host na Survey."

    - name: Validar host no inventario
      ansible.builtin.assert:
        that:
          - target_host in groups['all']
        fail_msg: "Host nao encontrado no inventario do START."

    - name: Consultar plataforma cadastrada
      ansible.builtin.set_fact:
        detected_platform: >-
          {{ hostvars[target_host].platform | default('unknown') | string | trim | lower }}

    - name: Validar plataforma suportada
      ansible.builtin.assert:
        that:
          - detected_platform in ['cisco_ios', 'cisco_asa']
        fail_msg: >-
          Plataforma nao identificada ou nao suportada
          para o host informado.

    - name: Publicar resultado no Artifact Store
      ansible.builtin.set_stats:
        data:
          target_host: "{{ target_host }}"
          platform: "{{ detected_platform }}"
          routing_status: "identified"
        per_host: false

    - name: Exibir resultado
      ansible.builtin.debug:
        msg:
          host: "{{ target_host }}"
          platform: "{{ detected_platform }}"
          routing_status: "identified"
