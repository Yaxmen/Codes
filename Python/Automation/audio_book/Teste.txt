
---
- name: ROUTE - Validar plataforma Cisco
  hosts: localhost
  gather_facts: false
  connection: local

  tasks:
    - name: Validar plataforma recebida
      ansible.builtin.assert:
        that:
          - platform is defined
          - expected_platform is defined
          - platform == expected_platform
        fail_msg: >-
          Rota nao selecionada.
          Plataforma identificada: {{ platform | default('unknown') }}.
          Rota esperada: {{ expected_platform | default('unknown') }}.
        success_msg: >-
          Rota selecionada: {{ expected_platform }}.
